4. Диаграмма классов

classDiagram
    class ArbiterSystem {
        +ShieldGateway gateway
        +SemanticRouter router
        +PresidioScanner scanner
        +PolicyEngine policy
        +process(request) Response
    }
    
    class SemanticRouter {
        -Encoder encoder
        -List routes
        -float threshold
        +classify(prompt) RouteResult
    }
    
    class Route {
        +string name
        +List utterances
        +ndarray centroid
    }
    
    class PresidioScanner {
        -AnalyzerEngine analyzer
        +analyze(text, lang) PIIResult
    }
    
    class PolicyEngine {
        -Dict policies
        +evaluate(context) Decision
    }
    
    ArbiterSystem *-- SemanticRouter
    ArbiterSystem *-- PresidioScanner
    ArbiterSystem *-- PolicyEngine
    SemanticRouter *-- Route

5. Жизненный цикл запроса

stateDiagram-v2
    [*] --> Received
    
    state Shield {
        Received --> RateCheck
        RateCheck --> JWTDecode: OK
        RateCheck --> RateLimited: Over
        JWTDecode --> AttackCheck
        AttackCheck --> Ready: Clean
        AttackCheck --> Blocked: Attack
    }
    
    state Classification {
        Ready --> SemanticRouter
        Ready --> PresidioScanner
        SemanticRouter --> WaitBoth
        PresidioScanner --> WaitBoth
    }
    
    state PolicyEval {
        WaitBoth --> Evaluate
        Evaluate --> Allowed
        Evaluate --> Denied
    }
    
    state Routing {
        Allowed --> InternalLLM
        Allowed --> ExternalLLM
        InternalLLM --> Response
        ExternalLLM --> Response
    }
    
    Denied --> BlockedResponse
    RateLimited --> ErrorResponse
    Blocked --> ErrorResponse
    Response --> AuditLog
    BlockedResponse --> AuditLog
    ErrorResponse --> AuditLog
    AuditLog --> [*]
